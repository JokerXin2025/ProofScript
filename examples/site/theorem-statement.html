    @theorem fermat_lemma_max
        (h_deriv : F.isDerivableAt x₀)
        (h_max : ∃ δ > 0, ∀ x ∈ Nbho x₀ δ, F.map x ≤ F.map x₀)
      : Deriv F x₀ 0
    := script
      obtain_exist ⟨L, h_L⟩ := h_deriv
      obtain_exist ⟨δ, h_δ_pos, h_max⟩ := h_max
      by suffices hzero : L = 0 by simpa [hzero] using h_L
      contra h_L_ne0
      · let q := fun x ↦ (f x - f x₀) / (x - x₀)
        by rcases lt_or_gt_of_ne h_L_ne0 with h_neg | h_pos
        · remark "If L < 0, all sufficiently nearby difference quotients are negative."
          obtain_exist ⟨r, h_r, h_q_neg⟩ := h_L.EventuallyNeg h_neg
          change ∀ x ∈ Nbhd x₀ r, q x < 0 at h_q_neg
          let t := min δ r / 2
          by have h_t_pos : t > 0 := by positivity
          by have h_t_le_δ : t < δ := by grind
          by have h_t_le_r : t < r := by grind
          remark "At a maximum, the left difference quotient is nonnegative."
          infer
            x₀ - t ∈ Nbho x₀ δ => f (x₀ - t) - f x₀ ≤ 0
                                  := sub_nonpos.mpr (h_max _ (by grind))
                              => (f (x₀ - t) - f x₀) / (x₀ - t - x₀) ≥ 0
                                  := div_nonneg_of_nonpos ?_ (by linarith)
                              => q (x₀ - t) ≥ 0
                                  := ?_
          by specialize h_q_neg (x₀ - t) (by grind)
          linarith
        · remark "If L > 0, all sufficiently nearby difference quotients are positive."
          obtain_exist ⟨r, h_r, h_q_pos⟩ := FuncLimit.EventuallyPos h_L h_pos
          change ∀ x ∈ Nbhd x₀ r, q x > 0 at h_q_pos
          let t := min δ r / 2
          by have h_t_pos : t > 0 := by positivity
          by have h_t_le_δ : t < δ := by grind
          by have h_t_le_r : t < r := by grind
          remark "At a maximum, the right difference quotient is nonpositive."
          infer
            x₀ + t ∈ Nbho x₀ δ => f (x₀ + t) - f x₀ ≤ 0
                                  := sub_nonpos.mpr (h_max _ (by grind))
                              => (f (x₀ + t) - f x₀) / (x₀ + t - x₀) ≤ 0
                                  := div_nonpos_of_nonpos_of_nonneg ?_ (by linarith)
                              => q (x₀ + t) ≤ 0
                                  := ?_
          by specialize h_q_pos (x₀ + t) (by grind)
          linarith
